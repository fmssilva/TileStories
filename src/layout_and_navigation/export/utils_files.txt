/**
 * NAVIGATION UTILITIES

 * Helper functions for navigation operations.
 * Pure functions - no side effects, just data transformation.

import { createElement } from 'react';
import { Route } from 'react-router-dom';
import type { NavItem } from '../types';
import type { Language } from '@/utils/language';

/**
 * Extract the correct language label from a NavItem
 * Handles both string and InlineTranslation formats
 * 
 * @param item - Navigation item with label
 * @param language - Current language ('pt' or 'en')
 * @param useShort - Whether to use labelShort if available
 * @returns Translated label string
 */
export function getNavItemLabel(
    item: NavItem,
    language: Language,
    useShort: boolean = false
): string {
    const labelField = useShort && item.labelShort ? item.labelShort : item.label;

    // Handle InlineTranslation object format
    if (typeof labelField === 'object' && labelField !== null) {
        return language === 'pt' ? labelField.pt : labelField.en;
    }

    // Handle plain string format (backwards compatibility)
    return labelField as string;
}

/**
 * Get navigation items for main menu
 * Filters by showInNav and sorts by order
 */
export function getMainNavItems(navConfig: NavItem[]): NavItem[] {
    return navConfig
        .filter(item => item.metadata?.showInNav)
        .sort((a, b) => (a.metadata?.order || 0) - (b.metadata?.order || 0));
}

/**
 * Get navigation items for footer
 * Same as main nav for now, but separate for flexibility
 */
export function getFooterNavItems(navConfig: NavItem[]): NavItem[] {
    return getMainNavItems(navConfig);
}

/**
 * Find navigation item by path
 * Supports exact match and pattern matching
 */
export function findNavItemByPath(navConfig: NavItem[], path: string): NavItem | null {
    // Try exact match first
    const exactMatch = navConfig.find(item => item.path === path);
    if (exactMatch) return exactMatch;

    // Try pattern matching for dynamic routes (e.g., /products/:id)
    for (const item of navConfig) {
        if (pathMatches(item.path, path)) {
            return item;
        }
        // Check children recursively
        if (item.children) {
            for (const child of item.children) {
                if (pathMatches(child.path, path)) {
                    return child;
                }
            }
        }
    }

    return null;
}

/**
 * Check if a route pattern matches a path
 * Simple pattern matching for :param style routes
 */
function pathMatches(pattern: string, path: string): boolean {
    if (pattern === path) return true;

    // Convert pattern to regex
    // /products/:id -> /products/[^/]+
    const regexPattern = pattern
        .replace(/:[^/]+/g, '[^/]+')
        .replace(/\*/g, '.*');

    const regex = new RegExp(`^${regexPattern}$`);
    return regex.test(path);
}

/**
 * Get all navigation items as flat array
 * Useful for searching/indexing
 */
export function getAllNavItems(navConfig: NavItem[]): NavItem[] {
    const allItems: NavItem[] = [];

    function addItems(items: NavItem[]) {
        for (const item of items) {
            allItems.push(item);
            if (item.children) {
                addItems(item.children);
            }
        }
    }

    addItems(navConfig);
    return allItems;
}

/**
 * Build breadcrumb trail from a navigation item
 * Walks up the parent chain to build the full path
 * 
 * Example: Product Detail -> Products -> Home
 */
export function buildBreadcrumbTrail(item: NavItem | null): NavItem[] {
    if (!item) return [];

    const trail: NavItem[] = [];
    let current: NavItem | undefined = item;

    // Walk up the parent chain
    while (current) {
        // Only include items that should show in breadcrumb
        if (current.metadata?.showInBreadcrumb !== false) {
            trail.unshift(current);
        }
        current = current.parent;
    }

    return trail;
}

/**
 * Populate parent references in navigation tree
 * Modifies items in place to add parent pointers
 * Call this once on app start to prepare navigation config
 */
export function populateParentReferences(items: NavItem[], parent?: NavItem): void {
    for (const item of items) {
        if (parent) {
            item.parent = parent;
        }
        if (item.children) {
            populateParentReferences(item.children, item);
        }
    }
}

/**
 * Check if a path is active
 * Supports exact and prefix matching
 */
export function isPathActive(currentPath: string, itemPath: string, exact = false): boolean {
    if (exact) {
        return currentPath === itemPath;
    }

    // Home route needs exact match to avoid matching everything
    if (itemPath === '/') {
        return currentPath === '/';
    }

    // For other routes, check if current path starts with item path
    return currentPath === itemPath || currentPath.startsWith(itemPath + '/');
}

/**
 * Extract parameter from dynamic route
 * 
 * Example:
 * extractParam('/products/123', '/products/:id', 'id') -> '123'
 */
export function extractParam(path: string, pattern: string, paramName: string): string | null {
    const patternParts = pattern.split('/');
    const pathParts = path.split('/');

    if (patternParts.length !== pathParts.length) {
        return null;
    }

    for (let i = 0; i < patternParts.length; i++) {
        if (patternParts[i] === `:${paramName}`) {
            return pathParts[i] || null;
        }
    }

    return null;
}

/**
 * Build path with parameters
 * 
 * Example:
 * buildPath('/products/:id', { id: '123' }) -> '/products/123'
 */
export function buildPath(pattern: string, params: Record<string, string>): string {
    let path = pattern;

    for (const [key, value] of Object.entries(params)) {
        path = path.replace(`:${key}`, value);
    }

    return path;
}

/**
 * Get navigation item depth in tree
 * 0 = top level, 1 = second level, etc.
 */
export function getItemDepth(item: NavItem): number {
    let depth = 0;
    let current: NavItem | undefined = item;

    while (current?.parent) {
        depth++;
        current = current.parent;
    }

    return depth;
}

/**
 * Find navigation item by ID
 * Searches entire tree recursively
 */
export function findNavItemById(items: NavItem[], id: string): NavItem | null {
    for (const item of items) {
        if (item.id === id) {
            return item;
        }
        if (item.children) {
            const found = findNavItemById(item.children, id);
            if (found) return found;
        }
    }
    return null;
}

/**
 * Get all sibling items (items at same level with same parent)
 */
export function getSiblings(item: NavItem, allItems: NavItem[]): NavItem[] {
    if (!item.parent) {
        // Top level items
        return allItems.filter(i => !i.parent);
    }

    return item.parent.children || [];
}

/**
 * Get next sibling item (for prev/next navigation)
 */
export function getNextSibling(item: NavItem, allItems: NavItem[]): NavItem | null {
    const siblings = getSiblings(item, allItems);
    const currentIndex = siblings.findIndex(i => i.id === item.id);

    if (currentIndex === -1 || currentIndex === siblings.length - 1) {
        return null;
    }

    return siblings[currentIndex + 1] || null;
}

/**
 * Get previous sibling item
 */
export function getPreviousSibling(item: NavItem, allItems: NavItem[]): NavItem | null {
    const siblings = getSiblings(item, allItems);
    const currentIndex = siblings.findIndex(i => i.id === item.id);

    if (currentIndex <= 0) {
        return null;
    }

    return siblings[currentIndex - 1] || null;
}

/**
 * Generate React Router Route elements from navigation config
 * 
 * This is the whole point of having a central navigation config!
 * - Eliminates manual route duplication in App.tsx
 * - Ensures routes always match navigation structure
 * - Reduces errors and maintenance burden
 * 
 * @param navConfig - Navigation configuration array
 * @returns Array of Route elements ready to render
 * 
 * @example
 * ```tsx
 * import { Routes, Route } from 'react-router-dom';
 * import { generateRoutesFromConfig } from '@/layout_and_navigation/utils';
 * import { navigationConfig } from '@/layout_and_navigation/config';
 * 
 * function App() {
 *   return (
 *     <Routes>
 *       {generateRoutesFromConfig(navigationConfig)}
 *     </Routes>
 *   );
 * }
 * ```
 */
export function generateRoutesFromConfig(navConfig: NavItem[]) {
    return navConfig.map((item) => {
        const Component = item.component;

        // Only generate route if component exists
        if (!Component) {
            return null;
        }

        // Generate Route element using imported Route and createElement
        return createElement(
            Route,
            {
                key: item.id,
                path: item.path,
                element: createElement(Component)
            }
        );
    }).filter(Boolean); // Remove null entries
}
