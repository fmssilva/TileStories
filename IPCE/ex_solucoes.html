x = altura que se pretende obter

t = q+s

Resolver o sistema com três equações e três incógnitas. Queremos obter o valor de "x" em função de "t".

x = 4.9 * (t^2 - 2*t*x/340 + (x/340)^2)

a = 4.9/340^2

import math

def discriminant(a: float, b: float, c: float) -> float:
    """ """
    return b*b - 4*a*c

def how_many_roots(a: float, b: float, c: float) -> int:
    """ """
    d = discriminant(a, b, c)
    if d < 0:
        return 0
    elif d == 0:
        return 1
    else:
        return 2

def root_left(a: float, b: float, c: float) -> float:
    """ """
    if a > 0:
        return (-b - math.sqrt(discriminant(a, b, c))) / (2*a)
    else:
        return (-b + math.sqrt(discriminant(a, b, c))) / (2*a)

def root_right(a: float, b: float, c: float) -> float:
    """ """
    if a > 0:
        return (-b + math.sqrt(discriminant(a, b, c)))/ (2*a)
    else:
        return (-b - math.sqrt(discriminant(a, b, c)))/ (2*a)

def main() -> None:
    a = float(input("A: "))
    b = float(input("B: "))
    c = float(input("C: "))
    n = how_many_roots(a, b, c)
    if n == 0:
        print("0")
    elif n == 1:
        print(f"1 {root_left(a,b,c)}")
    else:
        print(f"2 {root_left(a,b,c)} {root_right(a,b,c)}")

    signal = 1

from __future__ import annotations

class Rational:
    def __init__(self: Rational, num: int, den: int) -> None:
        self.num = num
        self.den = den
        self.simplify()
    
    def gcd(m: int, n: int) -> int:
        x = abs(m)
        y = abs(n)
        while x != y:
            if x > y:
                x -= y
            else:
                y -= x
        return x
    
    def simplify(self: Rational) -> Rational:
        common_divisor = Rational.gcd(self.num, self.den)
        self.num //= common_divisor
        self.den //= common_divisor
        if self.den < 0:
            self.num *= -1
            self.den *= -1
    
    def __str__(self: Rational) -> str:
        return f"{self.num}/{self.den}"
    
    def __add__(self: Rational, other: Rational) -> Rational:
        result = Rational(self.num * other.den + other.num * self.den, self.den * other.den)
        result.simplify()
        return result
    
    def __sub__(self: Rational, other: Rational) -> Rational:
        result = Rational(self.num * other.den - other.num * self.den, self.den * other.den)
        result.simplify()
        return result
    
    def __mul__(self: Rational, other: Rational) -> Rational:
        result = Rational(self.num * other.num, self.den * other.den)
        result.simplify()
        return result
    
    def __truediv__(self: Rational, other: Rational) -> Rational:
